    class LunaExplainer(DebugMixin, ABC):
    """
    Base class providing a unified explanation workflow in LUNA XAI.

    Notes
    -----
    Subclasses are expected to implement `generate_explanation_data()` to
    construct domain-specific neighborhood or sensitivity representations.

    Parameters
    ----------
    model : Any
        Predictive model implementing at least a `predict()` method.
    domain : Any
        Domain object responsible for preparing data and providing model access.
    method : Any
        Explanation/sensitivity method.
    metric : Any
        Metric evaluating the method's output.
    renderer : Any, optional
        Optional visualization component.
    output_dir : str, default="results"
        Output directory for artifacts generated by renderers.
    normalizer : Any, optional
        Optional importance-normalization function.
    debug : bool, default False
        If True, enables debug-level logging for this explainer instance.
    """

    def __init__(
        self,
        model: Any,
        domain: Any,
        method: Any,
        metric: Any,
        renderer: Optional[Any] = None,
        output_dir: str = "results",
        normalizer: Optional[Any] = None,
        debug: bool = False,
    ) -> None:
        self.model = model
        self.domain = domain
        self.method = method
        self.metric = metric
        self.renderer = renderer
        self.output_dir = output_dir
        self.normalizer = normalizer

        # Logger for LunaExplainer instances.
        # By default, application code is expected to configure handlers to
        # write logs into a file. If console output is preferred, the
        # "luna_xai" logger can be configured with a StreamHandler or via
        # `configure_luna_logging`.
        self._logger = get_luna_logger("explainer.luna")
        self._debug_enabled: bool = bool(debug)

    # ------------------------------------------------------------------
    # Stage 1 — Generate explanation data
    # ------------------------------------------------------------------
    def generate_explanation_data(self, instance: Any, **kwargs: Any) -> Dict[str, Any]:
        """
        Construct raw explanation data (predictions, neighborhood, sensitivities).

        Notes
        -----
        This method must be implemented by each concrete explainer variant.
        The method should *not* compute feature importances nor perform rendering.

        Returns
        -------
        dict
            {
                "sensitivity_data": {...},
                "predictions": array-like (optional),
                "metadata": {...}
            }
        """
        raise NotImplementedError(
            f"{self.__class__.__name__}.generate_explanation_data() must be implemented by subclasses."
        )

    # ------------------------------------------------------------------
    # Stage 2 — Compute importances and optionally render
    # ------------------------------------------------------------------
    def explain_from_data(
        self,
        explanation_data: Dict[str, Any],
        render: bool = True,
        context: Optional[Dict[str, Any]] = None,
        **kwargs: Any,
    ) -> MetricResult:
        """
        Compute feature importances and optionally render results using precomputed
        explanation data.

        Parameters
        ----------
        explanation_data : dict
            Output from `generate_explanation_data()`.
        render : bool, default=True
            Whether to produce visual output.
        context : dict, optional
            Metadata for the renderer (e.g., instance ID).

        Returns
        -------
        MetricResult
            Final evaluated explanation result.
        """
        self._debug(
            f"Explaining from precomputed data. "
            f"Keys={list(explanation_data.keys())}, render={render}."
        )

        if not explanation_data or "sensitivity_data" not in explanation_data:
            raise ValueError(
                "[LUNA] Missing or invalid explanation_data in explain_from_data()."
            )

        sensitivity_data = explanation_data["sensitivity_data"]

        # Metric evaluation
        self._debug("Computing metric from sensitivity data.")
        result = self.metric.compute(sensitivity_data)
        result.metadata.update(explanation_data.get("metadata", {}))

        # Optional normalization
        if self.normalizer is not None and result.importances:
            self._debug("Applying importance normalization function.")
            result.importances = self.normalizer(result.importances)

        # Optional rendering
        if render and self.renderer is not None:
            self._debug("Rendering explanation result with renderer component.")
            self.renderer.render(result, context=context or {})

        self._debug("Explanation from data completed successfully.")
        return result

    # ------------------------------------------------------------------
    # Stage 3 — Full pipeline
    # ------------------------------------------------------------------
    def explain_instance(self, instance: Any, **kwargs: Any) -> MetricResult:
        """
        One-step pipeline:
        generate explanation data → compute importances → (optional) render.

        This preserves backward compatibility with previous demos.
        """
        self._debug(
            f"Starting full explanation pipeline for instance of type "
            f"{type(instance).__name__}."
        )
        explanation_data = self.generate_explanation_data(instance, **kwargs)
        result = self.explain_from_data(
            explanation_data,
            render=True,
            context={"scope": "local"},
        )
        self._debug("Full explanation pipeline completed successfully.")
        return result

    # ------------------------------------------------------------------
    def __repr__(self) -> str:
        return (
            f"{self.__class__.__name__}("
            f"model={type(self.model).__name__}, "
            f"domain={type(self.domain).__name__}, "
            f"method={type(self.method).__name__}, "
            f"metric={type(self.metric).__name__}, "
            f"renderer={type(self.renderer).__name__ if self.renderer else 'None'})"
        )

    def auto_discover_core_components(verbose: bool = False) -> None:
    """
    Import all submodules under known core namespaces to trigger
    auto-registration via decorator pattern.

    Parameters
    ----------
    verbose : bool, default=False
        Whether to log warnings during discovery.
    """
    for base_pkg in CORE_MODULES:
        try:
            pkg = importlib.import_module(base_pkg)

            if not hasattr(pkg, "__path__"):
                # Not a package (should not happen in normal usage)
                continue

            for _, mod_name, _ in pkgutil.iter_modules(pkg.__path__):
                importlib.import_module(f"{base_pkg}.{mod_name}")

            if verbose:
                LOGGER.debug("Auto-discovered components under '%s'.", base_pkg)

        except Exception as exc:
            if verbose:
                LOGGER.warning(
                    "Could not load core package '%s' during auto-discovery: %s",
                    base_pkg,
                    exc,
                )

    def get_component(full_name: str) -> Type[Any]:
    """
    Retrieve a component by its fully qualified name: 'namespace.name'.

    Parameters
    ----------
    full_name : str
        Identifier of the component ('method.dsa', 'metric.ch', etc.)

    Returns
    -------
    type
        The class registered under the requested name.

    Raises
    ------
    KeyError
        If the component or namespace is missing.
    ValueError
        If the format is invalid.
    """
    if "." not in full_name:
        raise ValueError(
            f"Invalid component identifier '{full_name}'. Expected format 'namespace.name'."
        )

    namespace, name = full_name.split(".", maxsplit=1)
    namespace = namespace.strip().lower()
    name = name.strip().lower()

    try:
        cls = REGISTRY[namespace][name]
        return cls
    except KeyError as exc:
        available = list_registered(namespace).keys()
        LOGGER.error(
            "Component '%s' not found in registry. Available in namespace '%s': %s",
            full_name,
            namespace,
            list(available),
        )
        raise KeyError(
            f"Component '{full_name}' not found in registry. "
            f"Available: {available}"
        ) from exc

    def register_component(
    namespace: str,
    name: str,
    cls: Type[Any] | None = None,
    *,
    overwrite: bool = False,
) -> Callable[[Type[Any]], Type[Any]]:
    """
    Register a component under a given namespace and name.

    Supports both decorator and direct invocation usage.

    Parameters
    ----------
    namespace : str
        Logical group of components (e.g., 'metric', 'method', 'renderer').
    name : str
        Component identifier within the namespace.
    cls : type, optional
        If provided, the registration happens immediately.
        If omitted, the function returns a decorator.
    overwrite : bool, default=False
        Whether existing entries can be replaced. Useful in rapid prototyping
        but should be False in production for safety.

    Returns
    -------
    Callable
        A decorator if `cls` is None, otherwise returns the registered class.

    Raises
    ------
    ValueError
        If attempting to register a duplicate name without overwrite=True.
    """

    if not isinstance(namespace, str) or not isinstance(name, str):
        raise TypeError("Both 'namespace' and 'name' must be strings.")

    namespace = namespace.strip().lower()
    name = name.strip().lower()

    if not namespace or not name:
        raise ValueError("Both 'namespace' and 'name' must be non-empty strings.")

    def decorator(inner_cls: Type[Any]) -> Type[Any]:
        REGISTRY.setdefault(namespace, {})

        # Prevent accidental overriding unless explicitly allowed
        if not overwrite and name in REGISTRY[namespace]:
            raise ValueError(
                f"Component '{namespace}.{name}' is already registered "
                f"as {REGISTRY[namespace][name].__name__}. "
                f"Use overwrite=True to replace."
            )

        REGISTRY[namespace][name] = inner_cls

        # Structured debug logging for registry actions
        LOGGER.debug(
            "Registered component '%s.%s' → %s",
            namespace,
            name,
            inner_cls.__name__,
        )

        return inner_cls

    return decorator if cls is None else decorator(cls)

    class DSAMethod(BaseMethod):
    """
    Deterministic Sensitivity Analysis (DSA).

    Generates PDP/ICE-style local sensitivity curves by:
        1. Extracting the hybrid-space neighborhood around an instance
        2. Generating synthetic samples (empirical)
        3. For each feature:
            - Enumerating the real neighbor feature levels
            - Overwriting the feature column with each level
            - Predicting the model response over the full neighborhood

    For classification, when a `target_class` index is provided and the
    underlying model implements `predict_proba`, DSA operates on the
    probability of that class instead of raw predictions.
    """

    # ------------------------------------------------------------------
    # Initialization
    # ------------------------------------------------------------------
    def __init__(
        self,
        n_neighbors: int = 7,
        n_samples: int = 1000,
        sampler: str = "lhs",
        seed: Optional[int] = None,
        debug: bool = False,
    ) -> None:
        """
        Parameters
        ----------
        n_neighbors : int, default=7
            Number of real neighbors used to define the local region.
        n_samples : int, default=1000
            Number of synthetic samples drawn from the local neighborhood.
        sampler : {"lhs", ...}, default="lhs"
            Sampling strategy used in `create_synthetic_samples`.
        seed : int, optional
            Random seed for deterministic sampling. If None, sampling
            may be non-deterministic unless a seed is provided via kwargs.
        debug : bool, default=False
            If True, enables debug-level logging via BaseMethod/DebugMixin.
        """
        super().__init__(debug=debug)

        self.n_neighbors = n_neighbors
        self.n_samples = n_samples
        self.sampler = sampler
        self.seed = seed

        # These are bound at runtime (factory or explainer will set domain/neighbor_finder)
        self.domain: Optional[Any] = None
        self.neighbor_finder: Optional[Any] = None

    # ------------------------------------------------------------------
    # Private helper: prediction for explanation
    # ------------------------------------------------------------------
    def _predict_for_explanation(
        self,
        df: pd.DataFrame,
        target_class: Optional[int] = None,
    ) -> np.ndarray:
        """
        Centralizes prediction logic for DSA.

        - If `target_class` is provided and the underlying model supports
          `predict_proba`, returns the probability for that class.
        - Otherwise, falls back to `domain.predict(df)`.

        Parameters
        ----------
        df : pd.DataFrame
            Perturbed neighborhood data.
        target_class : int or None
            Index of the class of interest for classification explanations.

        Returns
        -------
        np.ndarray
            1D array of scalar responses (predictions or probabilities).
        """
        model = getattr(self.domain, "model", None)
        data_interface = getattr(self.domain, "data_interface", None)

        # Classification path: use predict_proba for the target class
        if target_class is not None and model is not None and hasattr(model, "predict_proba"):
            if data_interface is not None and hasattr(data_interface, "transform_input"):
                X_input = data_interface.transform_input(df)
            else:
                X_input = df

            if hasattr(X_input, "toarray"):
                X_input = X_input.toarray()

            X_input = np.asarray(X_input, dtype=np.float32, order="C")

            proba = model.predict_proba(X_input)

            if proba.ndim != 2 or target_class < 0 or target_class >= proba.shape[1]:
                raise RuntimeError(
                    "[DSA ERROR] Invalid target_class index or predict_proba output "
                    f"shape={proba.shape}, target_class={target_class}."
                )

            return np.asarray(proba[:, target_class], dtype=float)

        # Regression / generic path: delegate to domain.predict
        preds = self.domain.predict(df)
        return np.asarray(preds, dtype=float)

    # ------------------------------------------------------------------
    # Main sensitivity generator
    # ------------------------------------------------------------------
    def generate(
        self,
        domain: Any,
        instance: pd.Series | pd.DataFrame,
        data: Any = None,  # kept for interface compatibility
        **kwargs: Any,
    ) -> Dict[str, Dict[Any, list]]:
        """
        Generate local PDP/ICE-style sensitivity mappings.

        Parameters
        ----------
        domain : BaseDomain
            Provides model prediction (`predict`) and neighbor search
            (via `neighbor_finder`).
        instance : pd.Series or pd.DataFrame
            Query instance. Will be normalized into a single-row DataFrame.
        data : Any, optional
            Unused placeholder, kept only for backward compatibility.
        **kwargs :
            Optional overrides such as:
                - sampler="lhs" | ...
                - seed=123
                - n_neighbors, n_samples (if desired in the future)
                - target_class=<int> (for classification explanations)

        Returns
        -------
        dict
            Nested sensitivity structure: {feature -> {level -> [predictions...]}}
        """
        # Bind domain and neighbor_finder
        self.domain = domain
        self.neighbor_finder = getattr(domain, "neighbor_finder", None)

        if self.neighbor_finder is None:
            raise RuntimeError(
                "[DSA ERROR] No neighbor finder found. "
                "Call domain.build_index() before using DSA."
            )

        # Optional classification target class index
        target_class = kwargs.get("target_class", None)

        # Ensure a single-row DataFrame
        instance_df = ensure_dataframe(instance)

        # ---------------------------------------------------------------
        # Step 1 — Retrieve local neighborhood
        # ---------------------------------------------------------------
        self._logger.info("[LUNA DSA] Retrieving local neighborhood...")

        neighbors = self.neighbor_finder.find_neighbors(
            instance=instance_df,
            n_neighbors=self.n_neighbors,
        )

        if neighbors is None or len(neighbors) == 0:
            raise RuntimeError("[DSA ERROR] No neighbors found for this instance.")

        # ---------------------------------------------------------------
        # Step 2 — Synthetic samples
        # ---------------------------------------------------------------
        # Effective sampler and seed: kwargs override instance attributes
        effective_sampler = kwargs.get("sampler", self.sampler)
        effective_seed = kwargs.get("seed", self.seed)

        self._logger.info(
            "[LUNA DSA] Creating synthetic samples (method='%s', n_samples=%d, seed=%s)",
            effective_sampler,
            self.n_samples,
            str(effective_seed),
        )

        synthetic = create_synthetic_samples(
            neighbors,
            n_samples=self.n_samples,
            method=effective_sampler,
            seed=effective_seed,
        )

        # ---------------------------------------------------------------
        # Step 3 — Cleanup (remove auxiliary columns, ensure DataFrame)
        # ---------------------------------------------------------------
        def clean(df_any: Any) -> pd.DataFrame:
            df_local = df_any
            if isinstance(df_local, pd.DataFrame):
                if "distance" in df_local.columns:
                    df_local = df_local.drop(columns=["distance"]).copy()
            return df_local

        neighbors_clean = clean(neighbors)
        synthetic_clean = clean(synthetic)
        instance_df = ensure_dataframe(instance_df)
        instance_clean = clean(instance_df)

        full_neighborhood = pd.concat(
            [neighbors_clean, synthetic_clean, instance_clean],
            ignore_index=True,
        )

        self._logger.info(
            "[LUNA DSA] Base neighborhood assembled (%d samples).",
            len(full_neighborhood),
        )

        # ---------------------------------------------------------------
        # Step 4 — PDP/ICE-style sensitivity computation
        # ---------------------------------------------------------------
        sensitivity_data: Dict[str, Dict[Any, list]] = {}
        base_df = full_neighborhood.copy()

        for feature in base_df.columns:
            # Possible values = unique values from real neighbors only
            base_levels = neighbors_clean[feature].dropna().unique()

            try:
                feature_values = sorted(base_levels)
            except TypeError:
                # Fallback when values are not directly comparable
                feature_values = sorted(base_levels, key=lambda x: str(x))

            sensitivity_data[feature] = {}

            for val in feature_values:
                temp_df = base_df.copy()
                temp_df[feature] = val

                preds = self._predict_for_explanation(
                    temp_df,
                    target_class=target_class,
                )
                preds = np.asarray(preds).ravel()

                sensitivity_data[feature][val] = preds.tolist()

                self._debug(
                    f"[DSA PDP] feature={feature}, value={val} "
                    f"→ {len(preds)} predictions"
                )

        self._logger.info("[LUNA DSA] Sensitivity data generated successfully.")
        return sensitivity_data

    def generate(
    self,
    domain: Any,
    instance: pd.Series | pd.DataFrame,
    data: Any = None,  # kept for interface compatibility
    **kwargs: Any,
) -> Dict[str, Dict[Any, list]]:
    """
    Generate local PDP/ICE-style sensitivity mappings.

    Parameters
    ----------
    domain : BaseDomain
        Provides model prediction (`predict`) and neighbor search
        (via `neighbor_finder`).
    instance : pd.Series or pd.DataFrame
        Query instance. Will be normalized into a single-row DataFrame.
    data : Any, optional
        Unused placeholder, kept only for backward compatibility.
    **kwargs :
        Optional overrides such as:
            - sampler="lhs" | ...
            - seed=123
            - n_neighbors, n_samples (if desired in the future)
            - target_class=<int> (for classification explanations)

    Returns
    -------
    dict
        Nested sensitivity structure: {feature -> {level -> [predictions...]}}
    """
    # Bind domain and neighbor_finder
    self.domain = domain
    self.neighbor_finder = getattr(domain, "neighbor_finder", None)

    if self.neighbor_finder is None:
        raise RuntimeError(
            "[DSA ERROR] No neighbor finder found. "
            "Call domain.build_index() before using DSA."
        )

    # Optional classification target class index
    target_class = kwargs.get("target_class", None)

    # Ensure a single-row DataFrame
    instance_df = ensure_dataframe(instance)

    # ---------------------------------------------------------------
    # Step 1 — Retrieve local neighborhood
    # ---------------------------------------------------------------
    self._logger.info("[LUNA DSA] Retrieving local neighborhood...")

    neighbors = self.neighbor_finder.find_neighbors(
        instance=instance_df,
        n_neighbors=self.n_neighbors,
    )

    if neighbors is None or len(neighbors) == 0:
        raise RuntimeError("[DSA ERROR] No neighbors found for this instance.")

    # ---------------------------------------------------------------
    # Step 2 — Synthetic samples
    # ---------------------------------------------------------------
    # Effective sampler and seed: kwargs override instance attributes
    effective_sampler = kwargs.get("sampler", self.sampler)
    effective_seed = kwargs.get("seed", self.seed)

    self._logger.info(
        "[LUNA DSA] Creating synthetic samples (method='%s', n_samples=%d, seed=%s)",
        effective_sampler,
        self.n_samples,
        str(effective_seed),
    )

    synthetic = create_synthetic_samples(
        neighbors,
        n_samples=self.n_samples,
        method=effective_sampler,
        seed=effective_seed,
    )

    # ---------------------------------------------------------------
    # Step 3 — Cleanup (remove auxiliary columns, ensure DataFrame)
    # ---------------------------------------------------------------
    def clean(df_any: Any) -> pd.DataFrame:
        df_local = df_any
        if isinstance(df_local, pd.DataFrame):
            if "distance" in df_local.columns:
                df_local = df_local.drop(columns=["distance"]).copy()
        return df_local

    neighbors_clean = clean(neighbors)
    synthetic_clean = clean(synthetic)
    instance_df = ensure_dataframe(instance_df)
    instance_clean = clean(instance_df)

    full_neighborhood = pd.concat(
        [neighbors_clean, synthetic_clean, instance_clean],
        ignore_index=True,
    )

    self._logger.info(
        "[LUNA DSA] Base neighborhood assembled (%d samples).",
        len(full_neighborhood),
    )

    # ---------------------------------------------------------------
    # Step 4 — PDP/ICE-style sensitivity computation
    # ---------------------------------------------------------------
    sensitivity_data: Dict[str, Dict[Any, list]] = {}
    base_df = full_neighborhood.copy()

    for feature in base_df.columns:
        # Possible values = unique values from real neighbors only
        base_levels = neighbors_clean[feature].dropna().unique()

        try:
            feature_values = sorted(base_levels)
        except TypeError:
            # Fallback when values are not directly comparable
            feature_values = sorted(base_levels, key=lambda x: str(x))

        sensitivity_data[feature] = {}

        for val in feature_values:
            temp_df = base_df.copy()
            temp_df[feature] = val

            preds = self._predict_for_explanation(
                temp_df,
                target_class=target_class,
            )
            preds = np.asarray(preds).ravel()

            sensitivity_data[feature][val] = preds.tolist()

            self._debug(
                f"[DSA PDP] feature={feature}, value={val} "
                f"→ {len(preds)} predictions"
            )

    self._logger.info("[LUNA DSA] Sensitivity data generated successfully.")
    return sensitivity_data

    class PlotRenderer(BaseRenderer):
    """
    Renderer for LUNA XAI plots.

    This renderer delegates the actual plotting to utility functions in
    ``luna_xai.utils.plotting`` and uses a ``PlotExportStrategy`` to
    persist the generated figures (Matplotlib-based by default).

    Parameters
    ----------
    plot_type : {"importance", "line", "boxplot"}, default "importance"
        Default type of plot to render when calling :meth:`render`.
    output_dir : str, default "results"
        Base directory where plots will be saved. Subdirectories per
        instance (``idx``) are created automatically.
    exporter : PlotExportStrategy, optional
        Strategy object responsible for saving Matplotlib figures.
        If None, :class:`MatplotlibPdfExporter` is used.
    features : list of str, optional
        Subset of feature names to be plotted in line/boxplot modes.
        If None, all available features in the sensitivity data are used.
    top_k_importances : int, default 5
        Number of top features to highlight in the importance plot.
    debug : bool, default False
        If True, enables lightweight debug messages via ``_debug()``.

    Notes
    -----
    Logging for this renderer is integrated with the global LUNA logging
    infrastructure. By default, handlers are configured to log to a file
    (see ``configure_luna_logging``). If you prefer console logging, attach
    a ``logging.StreamHandler`` to the ``"luna_xai"`` logger instead.
    """

    VALID_PLOT_TYPES = {"importance", "line", "boxplot"}

    def __init__(
        self,
        plot_type: str = "importance",
        output_dir: str = "results",
        exporter: Optional[PlotExportStrategy] = None,
        features: Optional[List[str]] = None,
        top_k_importances: int = 5,
        debug: bool = False,
    ) -> None:
        plot_type_normalized = plot_type.lower()
        if plot_type_normalized not in self.VALID_PLOT_TYPES:
            raise ValueError(
                f"[PlotRenderer] Invalid plot_type '{plot_type}'. "
                f"Expected one of {sorted(self.VALID_PLOT_TYPES)}."
            )

        if top_k_importances <= 0:
            raise ValueError(
                "[PlotRenderer] 'top_k_importances' must be a positive integer."
            )

        # Initialize BaseRenderer / DebugMixin
        super().__init__(debug=debug)

        self.plot_type: str = plot_type_normalized
        self.output_dir: str = output_dir
        self.exporter: PlotExportStrategy = exporter or MatplotlibPdfExporter()
        self.features: Optional[List[str]] = features
        self.top_k_importances: int = int(top_k_importances)

        # Use a more specific logger name than the base "renderer"
        self._logger = get_luna_logger("renderer.plot")

        os.makedirs(self.output_dir, exist_ok=True)

    # ------------------------------------------------------------------
    def render(self, result: MetricResult, context: Optional[Dict[str, Any]] = None) -> None:
        """
        Render a plot from a :class:`MetricResult` object.

        Parameters
        ----------
        result : MetricResult
            Result object containing importances and/or sensitivity data.
        context : dict, optional
            Extra metadata used to customize the plot, typically including:

            - ``scope`` : {"local", "global"}, default "local"
            - ``instance`` : object, optional
            - ``idx`` : str, default "instance"
            - ``instance_prediction`` : float, optional

        Raises
        ------
        TypeError
            If ``result`` is not a :class:`MetricResult` instance.
        ValueError
            If the configured ``plot_type`` is invalid.
        """
        if not isinstance(result, MetricResult):
            raise TypeError(
                f"[PlotRenderer] Expected MetricResult, got {type(result).__name__}."
            )

        context = context or {}
        scope = context.get("scope", "local")
        instance = context.get("instance")
        idx = context.get("idx", "instance")
        instance_pred = context.get("instance_prediction")

        self._logger.info(
            "Rendering %s explanation as '%s' plot (idx=%s).",
            scope.upper(),
            self.plot_type,
            idx,
        )

        self._debug(
            f"Rendering {scope.upper()} explanation as '{self.plot_type}' plot "
            f"(idx={idx})."
        )

        if self.plot_type == "importance":
            self._render_importances(result, idx)
        elif self.plot_type == "line":
            self._render_lineplots(result.sensitivity_data, instance, idx)
        elif self.plot_type == "boxplot":
            self._render_boxplots(result.sensitivity_data, instance, idx, instance_pred)
        else:
            # Should be unreachable due to __init__ validation
            raise ValueError(f"[PlotRenderer] Invalid plot_type '{self.plot_type}'.")

    def render_from_data(
        self,
        sensitivity_data: Dict[str, Dict[Any, List[float]]],
        plot_type: Optional[str] = None,
        instance: Optional[Any] = None,
        idx: str = "instance",
        instance_prediction: Optional[float] = None,
    ) -> None:
        """
        Render plots directly from raw sensitivity data.

        Parameters
        ----------
        sensitivity_data : dict
            Nested dictionary of sensitivity values, typically in the same
            format produced by the sensitivity methods (e.g. DSA).
        plot_type : {"line", "boxplot"}, optional
            Plot mode to use. If None, falls back to the renderer's
            default ``plot_type``.
        instance : object, optional
            Instance associated with the sensitivity data.
        idx : str, default "instance"

        instance_prediction : float, optional

        - This method does **not** support ``"importance"``; that mode

            return

        effective_plot_type = (plot_type or self.plot_type).lower()
        self._logger.info(
            "Rendering raw sensitivity data as '%s' (idx=%s).",
            effective_plot_type,
            idx,
        )
        self._debug(
            f"Rendering raw sensitivity data as '{effective_plot_type}' "
            f"(idx={idx})."
        )

        if effective_plot_type == "line":
            self._render_lineplots(sensitivity_data, instance, idx)
        elif effective_plot_type == "boxplot":
            self._render_boxplots(sensitivity_data, instance, idx, instance_prediction)
        else:
            raise ValueError(
                f"[PlotRenderer] Unsupported plot_type '{effective_plot_type}' "
                "for render_from_data (only 'line' and 'boxplot' are allowed)."
            )

    # ------------------------------------------------------------------
    def _render_importances(self, result: MetricResult, idx: str) -> None:

        importances = result.importances

        if not importances:
            self._debug("[PlotRenderer] No importances to plot.")
            return

        self._debug(
            f"Rendering importance plot for idx={idx} with "
            f"{len(importances)} features (top_k={self.top_k_importances})."
        )

        fig = plot_importances(
            idx=idx,
            importances_dict=importances,
            type_="local",
            title="DSA",
            top_k=self.top_k_importances,
        )

        save_path = f"{self.output_dir}/{idx}/importances"
        os.makedirs(os.path.dirname(save_path), exist_ok=True)
        self.exporter.save(fig, save_path)

    # ------------------------------------------------------------------
    def _render_lineplots(
        self,
        sensitivity_data: Dict[str, Dict[Any, List[float]]],
        instance: Optional[Any],
        idx: str,
    ) -> None:
        """Internal helper to render line plots of feature trends."""
        self._debug(
            f"Rendering line plots for idx={idx}, "
            f"features={self.features if self.features is not None else 'ALL'}."

            idx=idx,
            sensitivity_data=sensitivity_data,
            instance=instance,
            features=self.features,

        os.makedirs(save_dir, exist_ok=True)

    # ------------------------------------------------------------------

    def render(self, result: MetricResult, context: Optional[Dict[str, Any]] = None) -> None:
    """
    Render a plot from a :class:`MetricResult` object.

    Parameters
    ----------
    result : MetricResult
        Result object containing importances and/or sensitivity data.
    context : dict, optional
        Extra metadata used to customize the plot, typically including:

        - ``scope`` : {"local", "global"}, default "local"
        - ``instance`` : object, optional
        - ``idx`` : str, default "instance"
        - ``instance_prediction`` : float, optional

    Raises
    ------
    TypeError
        If ``result`` is not a :class:`MetricResult` instance.
    ValueError
        If the configured ``plot_type`` is invalid.
    """
    if not isinstance(result, MetricResult):
        raise TypeError(
            f"[PlotRenderer] Expected MetricResult, got {type(result).__name__}."
        )

    context = context or {}
    scope = context.get("scope", "local")
    instance = context.get("instance")
    idx = context.get("idx", "instance")
    instance_pred = context.get("instance_prediction")

    self._logger.info(
        "Rendering %s explanation as '%s' plot (idx=%s).",
        scope.upper(),
        self.plot_type,
        idx,
    )

    self._debug(
        f"Rendering {scope.upper()} explanation as '{self.plot_type}' plot "
        f"(idx={idx})."
    )

    if self.plot_type == "importance":
        self._render_importances(result, idx)
    elif self.plot_type == "line":
        self._render_lineplots(result.sensitivity_data, instance, idx)
    elif self.plot_type == "boxplot":
        self._render_boxplots(result.sensitivity_data, instance, idx, instance_pred)
    else:
        # Should be unreachable due to __init__ validation
        raise ValueError(f"[PlotRenderer] Invalid plot_type '{self.plot_type}'.")

    def arctan_normalize(
    importances: Dict[str, float],
    scale_factor: float = 1000.0,
) -> Dict[str, float]:
    """
    Apply arctangent-based normalization with a scaling factor.

    This transformation smoothly compresses large-magnitude importance
    values into the interval [-1, 1], preserving sign and improving
    stability/interpretability. Values close to zero remain almost
    unchanged; very large magnitudes asymptotically approach -1 or 1.

    The transformation is defined as:

        norm(x) = arctan(x / scale_factor) / (π / 2)

    Parameters
    ----------
    importances : dict of str to float
        Raw importance scores from the metric. Values must be numeric
        (castable to ``float``).
    scale_factor : float, default 1000.0
        Controls curvature — larger values result in gentler compression
        (i.e., the arctan transition becomes slower).

    Returns
    -------
    dict of str to float
        Normalized importance scores in the range [-1, 1].

    Raises
    ------
    TypeError
        If ``importances`` is not a dict or if any value is not numeric.
    ValueError
        If ``scale_factor`` is not a positive float.
    """
    if not isinstance(importances, dict):
        raise TypeError(
            "arctan_normalize expects a dict[str, float] as 'importances', "
            f"got {type(importances).__name__}."
        )

    if scale_factor <= 0.0:
        raise ValueError(
            "Parameter 'scale_factor' must be a positive float; "
            f"received {scale_factor}."
        )

    _LOGGER.debug(
        "Starting arctan normalization for %d features (scale_factor=%g).",
        len(importances),
        scale_factor,
    )

    normalized: Dict[str, float] = {}

    for feature, value in importances.items():
        try:
            x = float(value)
        except (TypeError, ValueError) as exc:
            raise TypeError(
                f"Importance value for feature '{feature}' is not numeric: {value!r}."
            ) from exc

        # Normalized arctan with curvature control
        scaled = x / scale_factor
        atan_value = np.arctan(scaled)
        normalized_value = float(atan_value / (np.pi / 2))

        normalized[feature] = normalized_value

    _LOGGER.debug("Arctan normalization complete.")

    return normalized
